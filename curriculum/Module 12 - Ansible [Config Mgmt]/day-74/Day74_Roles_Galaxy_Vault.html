🔄 Ansible Galaxy
# Search for roles
ansible-galaxy search nginx

# Install a role
ansible-galaxy install geerlingguy.nginx,3.1.4

# Install from requirements.yml
ansible-galaxy install -r requirements.yml

# List installed roles
ansible-galaxy list

# Create a new role skeleton
ansible-galaxy init my_custom_role
📝 Jinja2 Templates

Template variables use {{ double_curly_braces }}:

# templates/nginx.conf.j2
server {
    listen {{ http_port }};
    server_name {{ server_name }};
    
    location / {
        root {{ document_root }};
        index index.html;
    }
}

# Using in tasks
- name: Deploy Nginx config
  ansible.builtin.template:
    src: nginx.conf.j2
    dest: /etc/nginx/nginx.conf
  notify: restart nginx
🔐 Ansible Vault
# Create encrypted file
ansible-vault create secrets.yml

# Encrypt existing file
ansible-vault encrypt vars/prod.yml

# View encrypted content
ansible-vault view secrets.yml

# Edit encrypted file
ansible-vault edit secrets.yml

# Run playbook with vault
ansible-playbook site.yml --ask-vault-pass

# Use vault password file (more secure for CI/CD)
ansible-playbook site.yml --vault-password-file ~/.vault_pass

Jagu's Rule: Never commit unencrypted secrets to Git! Use Vault for all sensitive data.
