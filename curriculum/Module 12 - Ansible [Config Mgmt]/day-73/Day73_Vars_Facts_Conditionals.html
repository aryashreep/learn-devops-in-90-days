🖥️ System Facts

Ansible automatically gathers facts about each managed node:

# View all facts
ansible all -m setup

# Filter specific facts
ansible all -m setup -a "filter=ansible_distribution*"

# Use facts in playbooks
- name: Print system info
  ansible.builtin.debug:
    msg: >
      OS: {{ ansible_distribution }} {{ ansible_distribution_version }}
      Architecture: {{ ansible_architecture }}
      Memory: {{ ansible_memtotal_mb }} MB
      CPUs: {{ ansible_processor_cores }}
🔀 Conditionals & Loops
# Conditional execution
- name: Install Apache on Debian only
  ansible.builtin.apt:
    name: apache2
    state: present
  when: ansible_os_family == "Debian"

# Loop over packages
- name: Install common tools
  ansible.builtin.apt:
    name: "{{ item }}"
    state: present
  loop:
    - htop
    - git
    - curl
    - vim

# Loop with loop_control
- name: Create users
  ansible.builtin.user:
    name: "{{ item }}"
    state: present
  loop:
    - alice
    - bob
  loop_control:
    pause: 2  # wait 2 seconds between iterations
🔔 Handlers
tasks:
  - name: Update Nginx config
    ansible.builtin.template:
      src: nginx.conf.j2
      dest: /etc/nginx/nginx.conf
    notify: restart nginx

  - name: Deploy new site
    ansible.builtin.copy:
      src: index.html
      dest: /var/www/html/index.html
    notify: reload nginx

handlers:
  - name: restart nginx
    ansible.builtin.service:
      name: nginx
      state: restarted

  - name: reload nginx
    ansible.builtin.service:
      name: nginx
      state: reloaded

Jagu's Tip: Handlers run ONCE at the end of the play, even if notified multiple times!
